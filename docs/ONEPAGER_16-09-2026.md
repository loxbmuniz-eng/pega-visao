# Volume da jornada: linhas acrescentadas, removidas e arquivos distintos
git log --since="2026-09-16 00:00" --numstat --pretty=format:"§%h" \
  | awk '/^§/{c++} /^[0-9]/{a+=$1;d+=$2;f[$3]=1} \
         END{print "commits:",c, "add:",a, "del:",d, "arquivos:",length(f)}'

# Arquivos criados na data (as guardas novas)

        --pretty=format:"" | grep -v '^$' | sort -u
